#include <iostream>
#include <chrono>

using namespace std;
using namespace std::chrono;

// Iterative Factorial
long long iterativeFactorial(int n)
{
    long long fact = 1;
    for (int i = 1; i <= n; i++)
    {
        fact *= i;
    }
    return fact;
}

// Recursive Factorial
long long recursiveFactorial(int n)
{
    if (n == 0 || n == 1)
        return 1;
    return n * recursiveFactorial(n - 1);
}

int main()
{
    int n;

    cout << "Enter a non-negative integer (e.g., 20): ";
    cin >> n;

    if (n < 0)
    {
        cout << "Factorial is not defined for negative numbers." << endl;
        return 0;
    }

    // Iterative Timing
    auto start1 = high_resolution_clock::now();
    long long iterativeResult = iterativeFactorial(n);
    auto stop1 = high_resolution_clock::now();

    auto iterativeTime = duration_cast<nanoseconds>(stop1 - start1);

    // Recursive Timing
    auto start2 = high_resolution_clock::now();
    long long recursiveResult = recursiveFactorial(n);
    auto stop2 = high_resolution_clock::now();

    auto recursiveTime = duration_cast<nanoseconds>(stop2 - start2);

    // Output
    cout << "\n--- Results for " << n << "! ---" << endl;

    cout << "Iterative Result : " << iterativeResult << endl;
    cout << "Iterative Time   : " << iterativeTime.count() << " ns" << endl;

    cout << "------------------------------" << endl;

    cout << "Recursive Result : " << recursiveResult << endl;
    cout << "Recursive Time   : " << recursiveTime.count() << " ns" << endl;

    return 0;
}
